    function doPost(e) {
  try {
    var requestData = JSON.parse(e.postData.contents);
    var htmlSnippet = requestData.html;
    if (!htmlSnippet) throw new Error("No HTML provided");

    var aiResult = null;
    for (var retry = 0; retry < 3; retry++) {
      try {
        aiResult = callOpenRouter(htmlSnippet);
        if (aiResult && aiResult.selectors) break;
      } catch (aiErr) {
        if (aiErr.message && aiErr.message.indexOf("429") !== -1) {
          Utilities.sleep((retry + 1) * 2000);
        }
        continue;
      }
    }

    if (!aiResult || !aiResult.selectors) {
      aiResult = heuristicParse(htmlSnippet);
    }

    if (!aiResult || !aiResult.selectors) {
      aiResult = getCachedConfig();
    }

    if (aiResult && aiResult.selectors) {
      cacheConfig(aiResult);
    }

    return ContentService.createTextOutput(JSON.stringify({
      success: !!(aiResult && aiResult.selectors),
      config: aiResult || {}
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: err.message
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function heuristicParse(html) {
  try {
    if (!html) return null;
    var config = { selectors: {}, sendIconPath: null };
    var inputMatch = html.match(/<[^>]*contenteditable[^>]*>.*?<\/[^>]*>/gi) || 
                     html.match(/<[^>]*aria-placeholder\s*=\s*"Message[^"]*"/gi);
    if (inputMatch && inputMatch.length > 0) {
      config.selectors.messageInput = 'div[contenteditable="true"], div[role="textbox"]';
    }
    if (html.indexOf('M22.513') !== -1 || html.indexOf('paper plane') !== -1) {
      config.sendIconPath = "M22.513 3.576C21.826 2.552";
      config.selectors.sendButton = 'div[role="button"] svg path[d^="M22.513"]';
    }
    if (!config.selectors.sendButton) {
      var btnMatch = html.match(/<[^>]*role\s*=\s*"button"[^>]*>.*?send.*?<\/[^>]*>/i) ||
                     html.match(/<[^>]*role\s*=\s*"button"[^>]*>[^<]{0,10}<\/[^>]*>/gi);
      if (btnMatch) config.selectors.sendButton = 'div[role="button"]';
    }
    if (config.selectors.messageInput || config.selectors.sendButton) return config;
  } catch (e) {}
  return null;
}

function cacheConfig(config) {
  try {
    var props = PropertiesService.getScriptProperties();
    props.setProperty('LAST_WORKING_CONFIG', JSON.stringify(config));
    props.setProperty('LAST_WORKING_CONFIG_TIMESTAMP', new Date().toISOString());
  } catch (e) {}
}
function getCachedConfig() {
  try {
    var props = PropertiesService.getScriptProperties();
    var cached = props.getProperty('LAST_WORKING_CONFIG');
    return cached ? JSON.parse(cached) : null;
  } catch (e) { return null; }
}

function callOpenRouter(html) {
  var apiKey = PropertiesService.getScriptProperties().getProperty("OPENROUTER_API_KEY");
  var url = "https://openrouter.ai/api/v1/chat/completions";
  var models = [
    "google/gemini-2.0-pro-exp-02-05:free",
    "google/gemini-2.0-flash-thinking-exp:free",
    "google/gemini-2.0-flash-exp:free",
    "meta-llama/llama-3.3-70b-instruct:free"
  ];
  var systemPrompt = "You are a specialized DOM-to-Selector API. You only output valid JSON.";
  var userPrompt = "Analyze this Instagram DM HTML snippet. Return JSON with selectors.messageInput, selectors.sendButton, and sendIconPath.";
  var lastError = null;
  for (var i = 0; i < models.length; i++) {
    try {
      var response = UrlFetchApp.fetch(url, {
        method: "post", contentType: "application/json",
        headers: { "Authorization": "Bearer " + apiKey },
        payload: JSON.stringify({
          model: models[i],
          messages: [
            { role: "system", content: systemPrompt },
            { role: "user", content: userPrompt + "\n\n" + html.substring(0, 5000) }
          ]
        }), muteHttpExceptions: true
      });
      var data = JSON.parse(response.getContentText());
      var content = data.choices[0].message.content;
      content = content.replace(/```json/g, '').replace(/```/g, '').trim();
      return JSON.parse(content);
    } catch (e) { lastError = e; }
  }
  throw new Error("All AI models failed: " + lastError.message);
}
